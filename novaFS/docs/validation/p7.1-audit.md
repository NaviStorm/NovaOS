# NovaFS P7.1 · Banc de non-écriture avant réparation

Ce programme utilise uniquement l’API publique du paquet P7 et le simulateur de stockage. Chaque dommage est injecté dans le support stable au repos. La sonde compte les appels de lecture par plage physique. Le journal du simulateur fournit séparément chaque tentative d’écriture et chaque synchronisation achevée. Les clés fixes du banc servent uniquement à déverrouiller les images jetables. Le moteur de production tire ses propres clés maîtresses et nonces.

Les commandes s’exécutent depuis la racine du monorepo sur un Mac Apple Silicon équipé de Rosetta et des deux cibles Rust Darwin. Elles compilent la bibliothèque locale et compilent le programme ci-dessous en deux exécutables. Sur la base `6ed50784ae73666caa15846ff380e5c7824faa36`, chacun termine avec le code **101** et quatre violations. Sur v0.9.10, chacun termine avec le code zéro et dix observations conformes par exécution. Les mesures de l’audit initial sont conservées en fin de page. [Traces des deux répétitions de qualification](v0.9.10/qualification.json.gz)

```sh
(
  set -eu
  export CARGO_TARGET_DIR="$PWD/novaFS/target"
  dest="$CARGO_TARGET_DIR/audit-p71"
  mkdir -p "$dest"
  awk '/^```rust$/{p=1;next} p && /^```$/{exit} p' \
    novaFS/docs/validation/p7.1-audit.md > "$dest/observation.rs"
  bilan=0
  for isa in arm64 x86_64; do
    case "$isa" in
      arm64) cible=aarch64-apple-darwin ;;
      x86_64) cible=x86_64-apple-darwin ;;
    esac
    NOVAFS_TARGET="$cible" novaFS/lab/run build --offline --lib
    lib="$CARGO_TARGET_DIR/$cible/release"
    sonde=
    if grep -q 'fn mutation_lock' novaFS/src/lib.rs; then sonde='--cfg=p71'; fi
    rustup run stable rustc ${sonde:+"$sonde"} --edition 2024 -O --target "$cible" \
      --extern "novafs=$lib/libnovafs.rlib" -L "dependency=$lib/deps" \
      "$dest/observation.rs" -o "$dest/observation-$isa"
    /usr/bin/arch "-$isa" "$dest/observation-$isa" || bilan=$?
  done
  exit "$bilan"
)
```

## Programme du banc

```rust
use novafs::crypto::{Credential, Secret};
use novafs::device::{Event, SimDevice, SimSnapshot};
use novafs::secure_pool::SecurePool;
use novafs::{BLOCK_BYTES, Compression, Device, Ids, Result};

use novafs::Block;
use std::{cell::Cell, rc::Rc};

#[derive(Default)]
struct Lectures {
    totaux: Cell<[u64; 4]>,
    avant_ecriture: Cell<Option<[u64; 4]>>,
}
struct Observe {
    support: SimDevice,
    mesures: Rc<Lectures>,
}
impl Device for Observe {
    #[cfg(p71)]
    fn mutation_lock(&self) -> Option<std::sync::Arc<std::sync::Mutex<()>>> {
        self.support.mutation_lock()
    }
    fn blocks(&self) -> u64 {
        self.support.blocks()
    }
    fn read(&self, bloc: u64) -> Result<Block> {
        let classe = match bloc {
            0..=1 => 0,
            2..=9 => 1,
            10..=45 => 2,
            _ => 3,
        };
        let mut n = self.mesures.totaux.get();
        n[classe] += 1;
        self.mesures.totaux.set(n);
        self.support.read(bloc)
    }
    fn write(&mut self, bloc: u64, page: &Block) -> Result<()> {
        if self.mesures.avant_ecriture.get().is_none() {
            self.mesures
                .avant_ecriture
                .set(Some(self.mesures.totaux.get()));
        }
        self.support.write(bloc, page)
    }
    fn flush(&mut self) -> Result<()> {
        self.support.flush()
    }
}

const POOL: [u8; 16] = [21; 16];

fn ouvrir<D: Device>(
    d: D,
    admin: &Secret,
    secours: &Secret,
    domaine: [u8; 16],
) -> Result<SecurePool<D>> {
    let mut p = SecurePool::open(d, POOL, admin)?;
    p.unlock(domaine, Credential::Recovery(secours))?;
    Ok(p)
}

fn verifier_image(
    image: &SimSnapshot,
    admin: &Secret,
    secours: &Secret,
    domaine: [u8; 16],
) -> Result<()> {
    ouvrir(image.device(), admin, secours, domaine)?.check()
}

#[derive(Clone, Copy, Debug, PartialEq)]
enum Cas {
    Temoin,
    Entete,
    BanqueEnReprise,
    BanqueEnEcriture,
    Donnee,
}

fn campagne(secteur: u32) -> Result<usize> {
    let admin = Secret::from_bytes([41; 32]);
    let secours = Secret::from_bytes([43; 32]);
    let d = SimDevice::new(1800, secteur)?;
    let ctl = d.control();
    let mut p = SecurePool::format(d, POOL, &admin)?;
    let v = p.create(
        [42; 16],
        Ids {
            fs: [45; 16],
            lv: [46; 16],
        },
        256,
        Compression::None,
        Some((b"fixture publique", &secours)),
    )?;
    let domaine = p.domain(v)?;
    p.create_file(v, "/f")?;
    p.write_file(v, "/f", 0, &[71; 8192])?;
    p.check()?;
    let g = p.generation();
    drop(p);
    let base = ctl.snapshot();
    let source = base.device();
    let mut corps = None;
    for lba in (46..source.blocks()).step_by(2) {
        let h = source.read(lba)?;
        if &h[..8] == b"NP7UNIT1"
            && h[11] == 1
            && h[32..48] == domaine
            && h[64..72] == g.to_le_bytes()
        {
            corps = Some(lba + 1);
            break;
        }
    }

    let mut violations = 0;
    for cas in [
        Cas::Temoin,
        Cas::Entete,
        Cas::BanqueEnReprise,
        Cas::BanqueEnEcriture,
        Cas::Donnee,
    ] {
        let d = base.device();
        let ctl = d.control();
        let certificat = 2 + 2 * (g % 4) + 1;
        if cas != Cas::BanqueEnEcriture {
            ctl.replace_stable(certificat, [0; BLOCK_BYTES])?;
        }
        let mesures = Rc::new(Lectures::default());
        let observe = Observe {
            support: d,
            mesures: Rc::clone(&mesures),
        };
        let mut p = ouvrir(observe, &admin, &secours, domaine)?;
        if cas == Cas::BanqueEnEcriture {
            p.advance()?;
        }
        p.check()?;
        let depart = p.generation();
        let cible = match cas {
            Cas::Temoin => None,
            Cas::Entete => Some((1, 80)),
            Cas::BanqueEnReprise | Cas::BanqueEnEcriture => Some((10 + 9 * (depart % 4) + 8, 0)),
            Cas::Donnee => Some((corps.expect("corps chiffré identifié"), 384)),
        };
        if let Some((bloc, octet)) = cible {
            let mut page = ctl.snapshot().device().read(bloc)?;
            page[octet] ^= 1;
            ctl.replace_stable(bloc, page)?;
        }
        let stable = ctl.snapshot();
        let avant = stable.bytes()?;
        let a_froid = verifier_image(&stable, &admin, &secours, domaine).is_ok();
        // Ne pas appeler check sur l'écrivain avant sa tentative de mutation :
        // une erreur le fermerait et masquerait le comportement propre de change.
        mesures.totaux.set([0; 4]);
        mesures.avant_ecriture.set(None);
        ctl.trace(None, None);
        let mutation = p.write_file(v, "/f", 0, b"nouveau");
        let evenements = ctl.events();
        let ecritures: Vec<_> = evenements
            .iter()
            .filter_map(|e| match e {
                Event::BeforeWrite(b) => Some(*b),
                _ => None,
            })
            .collect();
        let apres = ctl.snapshot();
        let identique = avant == apres.bytes()?;
        let remonte = verifier_image(&apres, &admin, &secours, domaine);
        let repare = cas != Cas::BanqueEnEcriture
            && apres.device().read(certificat)? == source.read(certificat)?;
        let flushs = evenements
            .iter()
            .filter(|e| matches!(e, Event::FlushCompleted(_)))
            .count();
        let attendu = if cas == Cas::Temoin {
            a_froid && mutation.is_ok() && repare && remonte.is_ok()
        } else {
            !a_froid && mutation.is_err() && ecritures.is_empty() && identique
        };
        println!(
            "{secteur} {cas:?} gen={depart}->{} controle_frais={a_froid} mutation={mutation:?} ecritures={} premier={:?} flushs={flushs} certificat_repare={repare} identique={identique} remonte={remonte:?} invariant={attendu} lectures_avant_ecriture_ou_refus={:?}",
            p.generation(),
            ecritures.len(),
            ecritures.first(),
            mesures
                .avant_ecriture
                .get()
                .unwrap_or_else(|| mesures.totaux.get())
        );
        violations += usize::from(!attendu);
    }
    Ok(violations)
}

fn main() -> Result<()> {
    let total = campagne(512)? + campagne(4096)?;
    assert_eq!(
        total, 0,
        "cas violant l’invariant de récupération sur cette architecture"
    );
    Ok(())
}
```
